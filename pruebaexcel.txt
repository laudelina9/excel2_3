        let preguntas = []; 
        let alumnoActual = "";
        let codigoActual = "";
        let temaActual = "";

        let ctxAudio = null;
        function initAudio() {
            if(!ctxAudio) { ctxAudio = new (window.AudioContext || window.webkitAudioContext)(); }
        }

        function reproducirSonidoBip(frecuencia, duracion) {
            try {
                initAudio();
                if(ctxAudio) {
                    let osc = ctxAudio.createOscillator();
                    let gain = ctxAudio.createGain();
                    osc.connect(gain);
                    gain.connect(ctxAudio.destination);
                    osc.frequency.setValueAtTime(frecuencia, ctxAudio.currentTime);
                    gain.gain.setValueAtTime(0.1, ctxAudio.currentTime);
                    osc.start();
                    osc.stop(ctxAudio.currentTime + duracion);
                }
            } catch(e) { console.log("Audio inactivo."); }
        }

        function activarPantallaCompleta() {
            let elem = document.documentElement;
            try {
                if (elem.requestFullscreen) { elem.requestFullscreen().catch(e => console.log("Salteado local.")); }
                else if (elem.webkitRequestFullscreen) { elem.webkitRequestFullscreen(); }
                else if (elem.msRequestFullscreen) { elem.msRequestFullscreen(); }
            } catch (err) { console.log("Bloqueo de navegador."); }
        }

        function intentarIngreso() {
            let txtNombre = document.getElementById("studentName").value.trim();
            let txtCodigo = document.getElementById("accessCode").value.trim().toUpperCase();
            let selTema = document.getElementById("examTema").value;

            if (!txtNombre || !txtCodigo || !selTema) {
                alert("ERR-01: Todos los campos, incluyendo la selección del Tema, son obligatorios.");
                return;
            }

            if (txtCodigo === CODIGO_DOCENTE) {
                esModoDocente = true;
                alumnoActual = "MODO_DOCENTE_TEST: " + txtNombre;
                codigoActual = txtCodigo;
                temaActual = selTema;
                inicializarExamen();
                return;
            }

            if (!CODIGOS_VALIDOS.includes(txtCodigo)) {
                alert("ERR-02: El código de terminal ingresado no es válido.");
                return;
            }

            if (localStorage.getItem(`status_${txtCodigo}`)) {
                alert("ACCESO DENEGADO: Esta terminal ya procesó una sesión evaluativa.");
                window.location.reload();
                return;
            }

            alumnoActual = txtNombre;
            codigoActual = txtCodigo;
            temaActual = selTema;
            inicializarExamen();
        }

        function inicializarExamen() {
            initAudio();
            reproducirSonidoBip(600, 0.15);
            activarPantallaCompleta(); 

            if (temaActual === "TEMA 1") {
                preguntas = JSON.parse(JSON.stringify(preguntasBase));
            } else {
                let primeraMitad = preguntasBase.slice(0, 10);
                let segundaMitad = preguntasBase.slice(10, 20);
                preguntas = JSON.parse(JSON.stringify(segundaMitad.concat(primeraMitad)));
            }
            
            document.getElementById("scrLogin").classList.remove("active");
            document.getElementById("scrGame").classList.add("active");

            // SE ACTIVA LA SÚPER GUARDIA AUTOMÁTICAMENTE SI NO ES DOCENTE
            if (!esModoDocente) { activarGuardiaSeguridad(); }
            cargarPregunta();
        }

        let itemsSeleccionadosMatcheo = { izq: null, der: null };

        function cargarPregunta() {
            let q = preguntas[indexPregunta];
            opcionTemporalSeleccionada = null; 
            
            document.getElementById("progBar").style.width = `${((indexPregunta) / preguntas.length) * 100}%`;
            document.getElementById("qNum").innerText = `${temaActual} ~ MÓDULO EVALUATIVO: PREGUNTA ${indexPregunta + 1} DE ${preguntas.length}`;
            document.getElementById("qText").innerText = q.texto;
            document.getElementById("qImg").src = q.img;

            let area = document.getElementById("answerArea");
            area.innerHTML = ""; 
            document.getElementById("btnNext").style.display = "none"; 

            if (q.tipo === "choice" || q.tipo === "concept") {
                let divGrid = document.createElement("div");
                divGrid.className = "options-grid";
                q.opciones.forEach((opt, idx) => {
                    let btn = document.createElement("button");
                    btn.className = "opt-btn";
                    btn.innerText = `${String.fromCharCode(65 + idx)}) ${opt}`;
                    btn.onclick = () => {
                        reproducirSonidoBip(450, 0.05);
                        let btns = divGrid.querySelectorAll(".opt-btn");
                        btns.forEach(b => b.classList.remove("selected"));
                        btn.classList.add("selected");
                        opcionTemporalSeleccionada = idx;
                        document.getElementById("btnNext").style.display = "block";
                    };
                    divGrid.appendChild(btn);
                });
                area.appendChild(divGrid);

            } else if (q.tipo === "combo") {
                let divCombo = document.createElement("div");
                divCombo.innerHTML = `<p style="font-size:0.9rem; color:var(--text-muted); margin-bottom:10px;">Despliegue el combo de selección y elija la opción técnica correcta:</p>`;

                optDef.innerText = "-- SELECCIONE UN CONCEPTO VÁLIDO --";
                select.appendChild(optDef);

                q.opciones.forEach((opt, idx) => {
                    let o = document.createElement("option");
                    o.value = idx;
                    o.innerText = opt;
                    select.appendChild(o);
                });

                select.onchange = () => {
                    reproducirSonidoBip(450, 0.05);
                    if(select.value !== "-1") {
                        opcionTemporalSeleccionada = parseInt(select.value);
                        document.getElementById("btnNext").style.display = "block";
                    } else {
                        opcionTemporalSeleccionada = null;
                        document.getElementById("btnNext").style.display = "none";
                    }
                };
                divCombo.appendChild(select);
                area.appendChild(divCombo);

            } else if (q.tipo === "match") {
                let divMatch = document.createElement("div");
                divMatch.innerHTML = `<p style="font-size:0.85rem; color:var(--accent-blue); margin-bottom:10px;">Instrucciones: Una todos los elementos cruzados.</p>`;
                
                let container = document.createElement("div");
                container.className = "match-container";

                let colIzq = document.createElement("div");
                colIzq.className = "match-col";
                colIzq.id = "mColIzq";

                let colDer = document.createElement("div");
                colDer.className = "match-col";
                colDer.id = "mColDer";

                let itemsIzq = q.parejas.map((p, i) => ({ text: p.izq, id: i })).sort(() => Math.random() - 0.5);
                let itemsDer = q.parejas.map((p, i) => ({ text: p.der, id: i })).sort(() => Math.random() - 0.5);

                itemsIzq.forEach(item => {
                    let d = document.createElement("div");
                    d.className = "match-item";
                    d.innerText = item.text;
                    d.dataset.id = item.id;
                    d.onclick = () => seleccionarMatchItem(d, "izq");
                    colIzq.appendChild(d);
                });

                itemsDer.forEach(item => {
                    let d = document.createElement("div");
                    d.className = "match-item";
                    d.innerText = item.text;
                    d.dataset.id = item.id;
                    d.onclick = () => seleccionarMatchItem(d, "der");
                    colDer.appendChild(d);
                });

                container.appendChild(colIzq);
                container.appendChild(colDer);
                divMatch.appendChild(container);
                area.appendChild(divMatch);

                q.respuestasTemporales = [];
            }
        }

        function seleccionarMatchItem(elemento, columna) {
            reproducirSonidoBip(500, 0.04);
            let hermanos = elemento.parentElement.querySelectorAll(".match-item");
            hermanos.forEach(h => h.classList.remove("selected"));

            elemento.classList.add("selected");
            itemsSeleccionadosMatcheo[columna] = elemento;

            if (itemsSeleccionadosMatcheo.izq && itemsSeleccionadosMatcheo.der) {
                let idIzq = itemsSeleccionadosMatcheo.izq.dataset.id;
                let idDer = itemsSeleccionadosMatcheo.der.dataset.id;
                let q = preguntas[indexPregunta];

                q.respuestasTemporales.push({ izqId: idIzq, derId: idDer, textoIzq: itemsSeleccionadosMatcheo.izq.innerText });

                itemsSeleccionadosMatcheo.izq.classList.remove("selected");
                itemsSeleccionadosMatcheo.izq.classList.add("linked");
                itemsSeleccionadosMatcheo.der.classList.remove("selected");
                itemsSeleccionadosMatcheo.der.classList.add("linked");

                itemsSeleccionadosMatcheo.izq = null;
                itemsSeleccionadosMatcheo.der = null;

                let restantes = document.querySelectorAll("#mColIzq .match-item:not(.linked)");
                if (restantes.length === 0) {
                    opcionTemporalSeleccionada = true; 
                    document.getElementById("btnNext").style.display = "block";
                }
            }
        }

        function siguientePregunta() {
            let q = preguntas[indexPregunta];

            if (opcionTemporalSeleccionada === null) {
                alert("Restricción: Es obligatorio responder para avanzar.");
                return;
            }

            let detalleUsuario = "";
            let respuestaCorrectaTexto = "";

            if (q.tipo === "choice" || q.tipo === "concept" || q.tipo === "combo") {
                detalleUsuario = q.opciones[opcionTemporalSeleccionada] || "Selección vacía";
                respuestaCorrectaTexto = q.opciones[q.correcta];
                
                respuestasAlumno.push({
                    preguntaOriginalId: q.id,
                    enunciado: q.texto,
                    tipo: q.tipo,
                    respuestaUsuario: detalleUsuario,
                    solucionCorrecta: respuestaCorrectaTexto,
                    esCorrecta: (opcionTemporalSeleccionada === q.correcta)
                });
            } else if (q.tipo === "match") {
                let aciertos = 0;
                q.respuestasTemporales.forEach(par => { if (par.izqId === par.derId) { aciertos++; } });
                
                respuestaCorrectaTexto = q.parejas.map(p => `[${p.izq} ➔ ${p.der}]`).join(" | ");

                respuestasAlumno.push({
                    preguntaOriginalId: q.id,
                    enunciado: q.texto,
                    tipo: q.tipo,
                    respuestaUsuario: "Ejercitación de emparejamiento completada",
                    solucionCorrecta: respuestaCorrectaTexto,
                    esCorrecta: (aciertos === q.parejas.length)
                });
            }

            indexPregunta++;
            if (indexPregunta < preguntas.length) {
                cargarPregunta();
            } else {
                finalizarYTransmitir(false);
            }
        }

        // --- SISTEMA COMPLETO DE SEGURIDAD ANTITRAMPAS ---
        function activarGuardiaSeguridad() {
            // 1. Detección por cambio de pestaña o minimizado
            window.onblur = function() { ejecutarBloqueoPorTrampa(); };
            document.onvisibilitychange = function() { if (document.visibilityState === 'hidden') { ejecutarBloqueoPorTrampa(); } };
            
            // 2. Detección obligatoria si achican o salen de la pantalla completa (Teclado o ESC)
            document.addEventListener("fullscreenchange", checkPantallaCompletaCambio);
            document.addEventListener("webkitfullscreenchange", checkPantallaCompletaCambio);
            document.addEventListener("mozfullscreenchange", checkPantallaCompletaCambio);
            document.addEventListener("MSFullscreenChange", checkPantallaCompletaCambio);

            // 3. Bloqueo manual de combinaciones críticas por teclado (Alt, Windows, F12)
            document.onkeydown = function(e) {
                if (e.key === "Escape" || e.keyCode === 27 || e.altKey || e.key === "Meta" || e.keyCode === 91 || e.key === "F12") {
                    ejecutarBloqueoPorTrampa();
                }
            };
        }

        function checkPantallaCompletaCambio() {
            // Si el estado de fullscreen pasa a ser nulo en medio del examen, significa que salieron
            let isFull = document.fullscreenElement || document.webkitFullscreenElement || document.mozFullScreenElement || document.msFullscreenElement;
            if (!isFull && !esModoDocente) {
                ejecutarBloqueoPorTrampa();
            }
        }

        function ejecutarBloqueoPorTrampa() {
            if (!trampaDetectada && !esModoDocente) {
                trampaDetectada = true;
                reproducirSonidoBip(220, 0.8);
                finalizarYTransmitir(true);
            }
        }
